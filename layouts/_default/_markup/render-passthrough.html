{{- /* Build-time math render hook. .Inner is the math source with delimiters already stripped by the passthrough extension; .Type is "inline" or "block". transform.ToMath (embedded KaTeX) renders it server-side. throwOnError fails the build on invalid LaTeX (org-math-lint should have caught it pre-publish; this is the backstop). */ -}} {{- $opts := dict "displayMode" (eq .Type "block") "throwOnError" true "output" "htmlAndMathml" -}} {{- with try (transform.ToMath .Inner $opts) -}} {{- with .Err -}} {{- errorf "render-passthrough: cannot render math in %q: %s" $.Page.File.Path . -}} {{- else -}} {{- .Value -}} {{- end -}} {{- end -}}
